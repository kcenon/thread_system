131 void enqueue(T value) {
132 if (shutdown_.load(std::memory_order_acquire)) {
133 return;
134 }
135
136 auto* new_node = new node{std::move(value)};
137
138 {
139 std::lock_guard<std::mutex> lock(tail_mutex_);
140 // Account for the item before publishing it: a consumer may remove
141 // it immediately after the release store (under a different lock).
142 size_.fetch_add(1, std::memory_order_release);
143 // Publish the initialized payload to the acquire load in try_dequeue().
144 // The separate head/tail mutexes cannot provide this ordering.
145 tail_->next.store(new_node, std::memory_order_release);
146 tail_ = new_node;
147 }
148
149 notify_one();
150 }

159 [[nodiscard]] auto try_dequeue() -> std::optional<T> {
160 std::lock_guard<std::mutex> lock(head_mutex_);
161
162 node* old_head = head_;
163 node* next = old_head->next.load(std::memory_order_acquire);
164
165 if (next == nullptr) {
166 return std::nullopt; // Queue is empty
167 }
168
169 // Extract value from next node
170 std::optional<T> value;
171 if (next->data.has_value()) {
172 value = std::move(*next->data);
173 next->data.reset();
174 }
175
176 // The acquired link publishes next's payload and completes the
177 // producer's last access to old_head, which can now be reclaimed.
178 head_ = next;
179 delete old_head;
180
181 size_.fetch_sub(1, std::memory_order_release);
182 return value;
183 }

191 template <typename Rep, typename Period>
192 [[nodiscard]] auto wait_dequeue(const std::chrono::duration<Rep, Period>& timeout)
193 -> std::optional<T> {
194 // First try without waiting
195 if (auto value = try_dequeue()) {
196 return value;
197 }
198
199 // Wait for notification
200 std::unique_lock<std::mutex> lock(cv_mutex_);
201 auto deadline = std::chrono::steady_clock::now() + timeout;
202
203 while (!shutdown_.load(std::memory_order_acquire)) {
204 // Try to dequeue
205 lock.unlock();
206 if (auto value = try_dequeue()) {
207 return value;
208 }
209 lock.lock();
210
211 // Check if we should stop waiting
212 if (std::chrono::steady_clock::now() >= deadline) {
213 return std::nullopt;
214 }
215
216 // Wait for notification or timeout
217 cv_.wait_until(lock, deadline);
218 }
219
220 // Final attempt after shutdown
221 return try_dequeue();
222 }
