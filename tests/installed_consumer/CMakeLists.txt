cmake_minimum_required(VERSION 3.20)
project(thread_installed_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(thread_system CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE thread_system::thread_system)
target_compile_definitions(consumer PRIVATE EXPECT_JTHREAD=${EXPECT_JTHREAD})
enable_testing()
add_test(NAME installed_worker COMMAND consumer)
set_tests_properties(installed_worker PROPERTIES TIMEOUT 60)

# Negative control: deliberately disagree with the installed library. The probe
# must reject this before any object with the wrong layout is allocated.
if(ABI_NEGATIVE)
  if(EXPECT_JTHREAD)
    if(MSVC)
      target_compile_options(consumer PRIVATE /UUSE_STD_JTHREAD)
    else()
      target_compile_options(consumer PRIVATE -UUSE_STD_JTHREAD)
    endif()
  else()
    target_compile_definitions(consumer PRIVATE USE_STD_JTHREAD)
  endif()
endif()
