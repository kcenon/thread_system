105 auto job_queue::enqueue(std::unique_ptr<job>&& value) -> common::VoidResult
106 {
107 // Early validation: check if queue is still accepting jobs
108 if (stop_.load())
109 {
110 return common::error_info{static_cast<int>(error_code::queue_stopped), "Job queue is stopped", "thread_system"};
111 }
112
113 // Validate input: null jobs are not allowed
114 if (value == nullptr)
115 {
116 return common::error_info{static_cast<int>(error_code::invalid_argument), "cannot enqueue null job", "thread_system"};
117 }
118
119 // Critical section: modify queue with proper synchronization
120 std::scoped_lock<std::mutex> lock(mutex_);
121
122 // Check size limit if bounded
123 if (max_size_.has_value() && queue_.size() >= max_size_.value())
124 {
125 return common::error_info{static_cast<int>(error_code::queue_full), "Job queue is full", "thread_system"};
126 }
127
128 // Move job into queue (efficient transfer of ownership)
129 queue_.push_back(std::move(value));
130 atomic_size_.fetch_add(1, std::memory_order_relaxed);
131
132 // Conditionally notify waiting consumers
133 if (notify_)
134 {
135 condition_.notify_one(); // Wake exactly one waiting thread
136 }
137
138 return common::ok();
139 }

194 auto job_queue::enqueue_batch(std::vector<std::unique_ptr<job>>&& jobs) -> common::VoidResult
195 {
196 // Early validation: check if queue is still accepting jobs
197 if (stop_.load())
198 {
199 return common::error_info{static_cast<int>(error_code::queue_stopped), "Job queue is stopped", "thread_system"};
200 }
201
202 // Validate batch: empty batches are not allowed
203 if (jobs.empty())
204 {
205 return common::error_info{static_cast<int>(error_code::invalid_argument), "cannot enqueue empty batch", "thread_system"};
206 }
207
208 // Pre-validate all jobs before modifying queue (ensures atomicity)
209 for (auto& job : jobs)
210 {
211 if (job == nullptr)
212 {
213 return common::error_info{static_cast<int>(error_code::invalid_argument), "cannot enqueue null job in batch", "thread_system"};
214 }
215 }
216
217 // Critical section: add entire batch with single lock acquisition
218 std::scoped_lock<std::mutex> lock(mutex_);
219
220 // Check size limit if bounded
221 if (max_size_.has_value())
222 {
223 std::size_t available_space = max_size_.value() - queue_.size();
224 if (jobs.size() > available_space)
225 {
226 return common::error_info{static_cast<int>(error_code::queue_full),
227 "Job queue cannot fit entire batch", "thread_system"};
228 }
229 }
230
231 // Move all jobs into queue efficiently
232 for (auto& job : jobs)
233 {
234 queue_.push_back(std::move(job));
235 }
236 atomic_size_.fetch_add(jobs.size(), std::memory_order_relaxed);
237
238 // Single notification for entire batch (performance optimization)
239 if (notify_)
240 {
241 condition_.notify_one(); // Wake one thread to process batch
242 }
243
244 return common::ok();
245 }

279 auto job_queue::dequeue() -> common::Result<std::unique_ptr<job>>
280 {
281 // Use unique_lock for condition variable operations
282 std::unique_lock<std::mutex> lock(mutex_);
283
284 // Block until job available OR queue stopped
285 condition_.wait(lock, [this]() { return !queue_.empty() || stop_.load(); });
286
287 // CRITICAL: Re-check queue state while still holding lock
288 // This prevents race with clear() that could empty the queue
289 // between wait() return and queue access (TOCTOU bug)
290 if (queue_.empty())
291 {
292 return common::error_info{static_cast<int>(error_code::queue_empty), "there are no jobs to dequeue", "thread_system"};
293 }
294
295 // Efficiently extract first job from queue
296 // At this point, we're guaranteed queue is not empty because:
297 // 1. We hold the mutex continuously from wait() return
298 // 2. We just verified !queue_.empty() above
299 auto value = std::move(queue_.front());
300 queue_.pop_front();
301 atomic_size_.fetch_sub(1, std::memory_order_relaxed);
302
303 return value; // Return moved job (caller takes ownership)
304 }

378 auto job_queue::dequeue_batch(void) -> std::deque<std::unique_ptr<job>>
379 {
380 std::deque<std::unique_ptr<job>> all_items;
381 {
382 // Critical section: atomically transfer all queue contents
383 std::scoped_lock<std::mutex> lock(mutex_);
384
385 // Efficient O(1) transfer of all jobs
386 std::swap(queue_, all_items);
387 atomic_size_.store(0, std::memory_order_relaxed);
388
389 // Wake all waiting threads since queue is now empty
390 condition_.notify_all();
391 }
392
393 return all_items; // Return all extracted jobs
394 }

427 auto job_queue::dequeue_batch_limited(std::size_t max_count)
428 -> std::deque<std::unique_ptr<job>>
429 {
430 std::deque<std::unique_ptr<job>> batch_items;
431 {
432 // Critical section: atomically extract up to max_count jobs
433 std::scoped_lock<std::mutex> lock(mutex_);
434
435 // Extract jobs up to max_count or queue size, whichever is smaller
436 std::size_t count = std::min(max_count, queue_.size());
437 for (std::size_t i = 0; i < count; ++i)
438 {
439 batch_items.push_back(std::move(queue_.front()));
440 queue_.pop_front();
441 }
442 atomic_size_.fetch_sub(count, std::memory_order_relaxed);
443
444 // Only notify if queue is now empty (to wake waiting workers)
445 // If jobs remain, other workers can continue dequeuing without wakeup overhead
446 if (queue_.empty())
447 {
448 condition_.notify_all();
449 }
450 }
451
452 return batch_items; // Return extracted batch
453 }

571 auto job_queue::get_memory_stats() const -> memory_stats
572 {
573 std::scoped_lock<std::mutex> lock(mutex_);
574
575 std::size_t job_count = queue_.size();
576
577 // Estimate memory usage
578 // std::unique_ptr<job> typically 8 bytes (pointer size)
579 // std::deque node overhead varies by platform, typically 32-48 bytes
580 constexpr std::size_t ptr_size = sizeof(std::unique_ptr<job>);
581 constexpr std::size_t node_overhead = 40; // Conservative estimate
582
583 // Use traditional aggregate initialization for C++17 compatibility
584 memory_stats stats;
585 stats.queue_size_bytes = (ptr_size + node_overhead) * job_count;
586 stats.pending_job_count = job_count;
587 stats.node_overhead_bytes = node_overhead * job_count;
588 return stats;
589 }

679 auto job_queue::inspect_pending_jobs(std::size_t limit) const
680 -> std::vector<diagnostics::job_info>
681 {
682 std::scoped_lock<std::mutex> lock(mutex_);
683
684 std::vector<diagnostics::job_info> result;
685 auto count = (limit == 0) ? queue_.size() : std::min(limit, queue_.size());
686 result.reserve(count);
687
688 auto now = std::chrono::steady_clock::now();
689 std::size_t index = 0;
690
691 for (const auto& job_ptr : queue_)
692 {
693 if (limit > 0 && index >= limit)
694 {
695 break;
696 }
697
698 if (job_ptr == nullptr)
699 {
700 ++index;
701 continue;
702 }
703
704 diagnostics::job_info info;
705 info.job_id = job_ptr->get_job_id();
706 info.job_name = job_ptr->get_name();
707 info.status = diagnostics::job_status::pending;
708 info.enqueue_time = job_ptr->get_enqueue_time();
709 info.start_time = job_ptr->get_enqueue_time(); // Not started yet
710
711 // Calculate wait time (time since enqueue)
712 info.wait_time = std::chrono::duration_cast<std::chrono::nanoseconds>(
713 now - job_ptr->get_enqueue_time()
714 );
715 info.execution_time = std::chrono::nanoseconds{0}; // Not started yet
716
717 result.push_back(std::move(info));
718 ++index;
719 }
720
721 return result;
722 }
